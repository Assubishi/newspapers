%%writefile batch_ocr_newspapers.py
import argparse
import shutil
import subprocess
import zipfile
from pathlib import Path

import fitz


def run(cmd):
    print("\n>>> " + " ".join(str(x) for x in cmd), flush=True)
    subprocess.run(cmd, check=True)


def count_pages(pdf_path: Path) -> int:
    doc = fitz.open(str(pdf_path))
    pages = len(doc)
    doc.close()
    return pages


def unzip_if_needed(zip_path: Path, input_dir: Path):
    input_dir.mkdir(parents=True, exist_ok=True)

    existing_pdfs = list(input_dir.rglob("*.pdf"))
    if existing_pdfs:
        print(f"PDFs already extracted: {len(existing_pdfs)}")
        return

    print(f"Unzipping {zip_path} -> {input_dir}")
    with zipfile.ZipFile(zip_path, "r") as z:
        z.extractall(input_dir)


def safe_name(path: Path) -> str:
    return path.stem.replace(" ", "_").replace("/", "_").replace("\\", "_")


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--zip", default="newspapers20.zip")
    parser.add_argument("--input-dir", default="/content/input_pdfs")
    parser.add_argument("--work-root", default="/content/ocr_work")
    parser.add_argument("--final-txt-dir", default="/content/final_txt")
    parser.add_argument("--dpi", type=int, default=300)
    parser.add_argument("--workers", type=int, default=2)
    parser.add_argument("--device", default="gpu:0")
    parser.add_argument("--lang", default="kaz+eng")
    parser.add_argument("--limit", type=int, default=0, help="Use 1 or 2 for testing. 0 = all PDFs.")
    parser.add_argument("--skip-existing", action="store_true")
    parser.add_argument("--no-furniture-clean", action="store_true")
    args = parser.parse_args()

    zip_path = Path(args.zip)
    input_dir = Path(args.input_dir)
    work_root = Path(args.work_root)
    final_txt_dir = Path(args.final_txt_dir)

    if not zip_path.exists():
        raise FileNotFoundError(f"ZIP not found: {zip_path.resolve()}")

    unzip_if_needed(zip_path, input_dir)

    work_root.mkdir(parents=True, exist_ok=True)
    final_txt_dir.mkdir(parents=True, exist_ok=True)

    pdfs = sorted(input_dir.rglob("*.pdf"))
    if args.limit > 0:
        pdfs = pdfs[:args.limit]

    print(f"PDFs to process: {len(pdfs)}")

    for i, pdf in enumerate(pdfs, start=1):
        name = safe_name(pdf)
        pdf_work_dir = work_root / name
        final_txt = final_txt_dir / f"{name}.txt"

        if args.skip_existing and final_txt.exists():
            print(f"[{i}/{len(pdfs)}] Skipping existing: {final_txt}")
            continue

        pages = count_pages(pdf)
        print(f"\n========== [{i}/{len(pdfs)}] {pdf.name} | pages={pages} ==========")

        run([
            "python",
            "run_hybrid_fast_reuse.py",
            "--pdf", str(pdf),
            "--pages", str(pages),
            "--dpi", str(args.dpi),
            "--workers", str(args.workers),
            "--device", args.device,
            "--lang", args.lang,
            "--tesseract", "/usr/bin/tesseract",
            "--output-root", str(pdf_work_dir),
        ])

        if args.no_furniture_clean:
            source_txt = pdf_work_dir / "all_pages_tsv_filtered.txt"
        else:
            run([
                "python",
                "remove_page_furniture_preserve_blanks.py",
                "--root", str(pdf_work_dir),
                "--pages", str(pages),
            ])
            source_txt = pdf_work_dir / "all_pages_no_furniture.txt"

        if not source_txt.exists():
            raise FileNotFoundError(f"Expected TXT not found: {source_txt}")

        shutil.copyfile(source_txt, final_txt)
        print(f"Saved: {final_txt}")

    zip_base = "/content/final_txt_results"
    shutil.make_archive(zip_base, "zip", final_txt_dir)
    print(f"\nDONE: {zip_base}.zip")


if __name__ == "__main__":
    main()
